🚀 Complete Integration Code

Here's how to implement push notifications with proper UI triggers using our SDK:

<!-- Step 1: Include our SDK --> <script src="https://pushads123.com/sdk.js"></script> <!-- Step 2: Add a trigger button --> <button id="enablePush">Enable Notifications 🔔</button> <script> // Step 3: Initialize SDK (doesn't request permission) PN.init({ apiKey: 'YOUR_API_KEY', baseUrl: 'https://pushads123.com' }); // Step 4: Handle user click (requests permission) document.getElementById("enablePush").addEventListener("click", async () => { try { // This will request permission and subscribe const result = await PN.subscribe(); if (result.success) { // Update UI to show success document.getElementById("enablePush").innerHTML = "✅ Notifications Enabled"; document.getElementById("enablePush").disabled = true; // Optional: Send test notification await fetch("https://pushads123.com/api/customer/notify", { method: "POST", headers: { "Content-Type": "application/json", "Authorization": "Bearer YOUR_API_KEY" }, body: JSON.stringify({ title: "Welcome! 🎉", body: "You'll now receive important updates", url: window.location.href }) }); } } catch (error) { console.error("Subscription failed:", error); alert(error.message); } }); // Step 5: Check if already subscribed on page load window.addEventListener('load', async () => { const status = await PN.getSubscriptionStatus(); if (status.isSubscribed) { document.getElementById("enablePush").innerHTML = "✅ Notifications Enabled"; document.getElementById("enablePush").disabled = true; } }); </script>
